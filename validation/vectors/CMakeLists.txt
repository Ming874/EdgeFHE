cmake_minimum_required(VERSION 3.16)
project(edgefhe_v1_openfhe_vectors LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenFHE CONFIG REQUIRED)
set(CMAKE_CXX_FLAGS "${OpenFHE_CXX_FLAGS}")
add_executable(export_vectors export_vectors.cpp)
target_include_directories(export_vectors PRIVATE ${OpenFHE_INCLUDE} ${OpenFHE_INCLUDE}/core ${OpenFHE_INCLUDE}/third-party/include)
target_link_directories(export_vectors PRIVATE ${OpenFHE_LIBDIR})
target_link_libraries(export_vectors PRIVATE ${OpenFHE_SHARED_LIBRARIES})
set_target_properties(export_vectors PROPERTIES BUILD_RPATH "${OpenFHE_LIBDIR}")
