cmake_minimum_required(VERSION 3.20)
project(EdgeFHE_openfhe_client_baseline LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenFHE CONFIG REQUIRED)
find_package(OpenMP REQUIRED)
set(CMAKE_CXX_FLAGS "${OpenFHE_CXX_FLAGS}")
add_executable(openfhe_client_baseline baseline.cpp)
target_include_directories(openfhe_client_baseline PRIVATE ${OpenFHE_INCLUDE} ${OpenFHE_INCLUDE}/core ${OpenFHE_INCLUDE}/pke ${OpenFHE_INCLUDE}/binfhe ${OpenFHE_INCLUDE}/third-party/include)
target_link_directories(openfhe_client_baseline PRIVATE ${OpenFHE_LIBDIR})
target_link_libraries(openfhe_client_baseline PRIVATE ${OpenFHE_SHARED_LIBRARIES} OpenMP::OpenMP_CXX)
set_target_properties(openfhe_client_baseline PROPERTIES BUILD_RPATH "${OpenFHE_LIBDIR}")
if(UNIX AND NOT APPLE)
  # DT_RPATH also finds OpenFHE's transitive shared dependencies (e.g. binfhe).
  target_link_options(openfhe_client_baseline PRIVATE "-Wl,--disable-new-dtags")
endif()
