cmake_minimum_required(VERSION 3.20)
project(EdgeFHE_Native_CPP_E2E LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This candidate targets Linux/WSL; no Windows network acceptance yet")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(EFNET_SANITIZE "Instrument the new transport and cloud with ASan/UBSan" OFF)
option(EFNET_TSAN "Instrument the new transport and cloud with TSan" OFF)
find_package(OpenSSL 3 REQUIRED)
find_package(Threads REQUIRED)
add_library(efnet_cloud SHARED vendor/application/native/cloud.cpp)
target_link_libraries(efnet_cloud PRIVATE Threads::Threads)
foreach(name server probe wire_selftest wire_fixture span_selftest permit_selftest)
  add_executable(efnet_${name} ${name}.cpp)
  target_link_libraries(efnet_${name} PRIVATE OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
endforeach()
target_link_libraries(efnet_server PRIVATE efnet_cloud)
target_link_libraries(efnet_wire_selftest PRIVATE efnet_cloud)
if(APP_BUILD)
  foreach(name owner normalize)
    find_library(EF_${name} NAMES ef_app_${name} PATHS "${APP_BUILD}" NO_DEFAULT_PATH REQUIRED)
  endforeach()
  add_executable(efnet_client client.cpp)
  target_link_libraries(efnet_client PRIVATE ${EF_owner} ${EF_normalize} OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
  set_target_properties(efnet_client PROPERTIES BUILD_RPATH "${APP_BUILD}")
endif()
get_property(all_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(target ${all_targets})
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
  if(EFNET_SANITIZE)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
  if(EFNET_TSAN)
    target_compile_options(${target} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=thread)
  endif()
endforeach()
enable_testing()
add_test(NAME wire_padding_scalar_oracle_and_refusals COMMAND efnet_wire_selftest)
