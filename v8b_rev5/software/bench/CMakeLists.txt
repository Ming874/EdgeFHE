cmake_minimum_required(VERSION 3.16)
project(EdgeFHE_Verification_Suite LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL 3 REQUIRED)
find_package(Threads REQUIRED)
# VENDOR = the accepted candidate's vendor tree (device header, V8b specification model, driver ABI/policy headers).
if(NOT VENDOR)
  message(FATAL_ERROR "-DVENDOR=<candidate/vendor>")
endif()
add_executable(suite_device suite_device.cpp)
target_include_directories(suite_device PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_include_directories(suite_device SYSTEM PRIVATE ${VENDOR}/application/native ${VENDOR}/frozen_sdk/src/client)
target_link_libraries(suite_device PRIVATE OpenSSL::Crypto Threads::Threads)
# The owner profiler links the accepted pinned owner libraries (APP_BUILD = directory with libef_app_owner.so).
if(APP_BUILD)
  find_library(EF_owner NAMES ef_app_owner PATHS "${APP_BUILD}" NO_DEFAULT_PATH REQUIRED)
  add_executable(suite_profile suite_profile.cpp)
  target_include_directories(suite_profile PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(suite_profile PRIVATE ${EF_owner} OpenSSL::Crypto Threads::Threads)
  set_target_properties(suite_profile PROPERTIES BUILD_RPATH "${APP_BUILD}")
endif()
foreach(t suite_device suite_profile)
  if(TARGET ${t})
    target_compile_options(${t} PRIVATE -O2 -Wall -Wextra -Werror -fno-fast-math)
  endif()
endforeach()
