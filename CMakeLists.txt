cmake_minimum_required(VERSION 3.20)
project(EdgeFHEFixedProfile VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The native backend uses Linux ioctl/mmap APIs. Use Linux or WSL; Windows Python checks remain available.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
function(edgefhe_target target source)
  add_executable(${target} ${source})
  target_link_libraries(${target} PRIVATE Threads::Threads)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
endfunction()
edgefhe_target(paired_public_vectors software/reference/paired_public_vectors.cpp)
edgefhe_target(mixed_runner software/reference/mixed_runner.cpp)
edgefhe_target(soak_runner software/board_tests/soak_runner.cpp)
edgefhe_target(concurrent_runner software/board_tests/concurrent_runner.cpp)
edgefhe_target(usercopy_runner software/board_tests/usercopy_runner.cpp)
add_library(edgefhe_backend SHARED software/ckks/backend.cpp)
target_link_libraries(edgefhe_backend PRIVATE Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(edgefhe_backend PRIVATE -Wall -Wextra -Werror)
endif()
include(CTest)
if(BUILD_TESTING)
  add_test(NAME public_cpu_exact COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/require_pass.py --marker "CPU_OPENFHE_EXACT_PASS cases=16 coefficients=196608" -- $<TARGET_FILE:paired_public_vectors> ${CMAKE_CURRENT_SOURCE_DIR}/validation/vectors/vectors.bin --cpu-check)
  add_test(NAME mixed_cpu_exact COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/require_pass.py --marker "MIXED_CPU_ORACLE_PASS cases=25 coefficients=307200" -- $<TARGET_FILE:mixed_runner> ${CMAKE_CURRENT_SOURCE_DIR}/validation/vectors/mixed_vectors.bin --cpu-check)
  add_test(NAME ckks_cpu_affine COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/require_pass.py --marker "HOST_CKKS_AFFINE_REFERENCE_PASS cases=6" -- ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/software/ckks/local_reference.py --library $<TARGET_FILE:edgefhe_backend>)
  set_tests_properties(ckks_cpu_affine PROPERTIES TIMEOUT 300)
  set_tests_properties(public_cpu_exact mixed_cpu_exact ckks_cpu_affine PROPERTIES LABELS "cpu-only" ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1;PYTHONUNBUFFERED=1" RUN_SERIAL TRUE)
endif()
message(STATUS "Build only: no module insertion, FPGA loading, networking or DMA. CTest is CPU-only.")

option(EDGEFHE_BUILD_NATIVE_CANDIDATE "Build locally reviewed native CKKS candidate" ON)
if(EDGEFHE_BUILD_NATIVE_CANDIDATE)
  add_subdirectory(software/native_ckks)
endif()

option(EDGEFHE_BUILD_V5 "Build the V5 backend/pipeline layer; host tests use a simulated DMA device" ON)
if(EDGEFHE_BUILD_V5)
  add_subdirectory(software/native_v5)
endif()
